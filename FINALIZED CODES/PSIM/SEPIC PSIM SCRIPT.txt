Clear();
Print("--- SEPIC script starting ---");

// SEPIC level-2 model.
// Runs the PSIM schematic at nine operating points and exports one TXT
// (fine timestep, used for switching-loss integration in Python) and one
// SMV (coarse timestep, used only for looking at waveforms) per case.
// Duty cycles come from the SEPIC v9.4 MATLAB model with the IRLZ44NPbF
// (tr=84ns, tf=15ns, RJA=62 C/W, dVC1=2%).

BasePath  = "C:/Users/User/Desktop/SEPIC NEW/";
SchFile   = BasePath + "FINAL SEPIC .psimsch";
IndexFile = BasePath + "SEPIC_case_index.csv";

// Gate resistor as fitted in the driver circuit
Rg = 3.4;

// Nine operating points, ordered Vin = 9, 12, 15 V with Io = 0.5, 1.0, 1.5 A
// at each (Vo = 12 V, so Rload = 24, 12, 8 ohm).
Vin_list   = { 9,  9,  9,  12, 12, 12,  15, 15, 15 };
Rload_list = { 24, 12, 8,  24, 12, 8,   24, 12, 8  };

// Duty cycles from the MATLAB model, matched one-to-one with the rows above.
Duty_list = {
    0.5836,  0.58685, 0.59069,   // Vin = 9 V
    0.5114,  0.51349, 0.5159,    // Vin = 12 V
    0.4552,  0.45669, 0.4584     // Vin = 15 V
};

FileWrite(IndexFile, "Case,Vin_V,Io_target_A,Rload_Ohm,Duty,Rg_Ohm,Waveform_File" + _CRLF, "A");

i = 0;
while (i < 9)
{
    Vin_case   = Vin_list[i];
    Rload_case = Rload_list[i];
    Duty_case  = Duty_list[i];
    Io_target  = 12 / Rload_case;
    CaseNo     = String(i + 1);

    TxtFile = BasePath + "SEPIC_case_" + CaseNo + "_steady_state.txt";
    SmvFile = BasePath + "SEPIC_case_" + CaseNo + "_steady_state.smv";

    Print("Case ", CaseNo, ": Vin=", Vin_case, "V, Rload=", Rload_case, "Ohm, Duty=", Duty_case, ", Rg=", Rg, "Ohm");

    // Fine timestep for the data file. The 15 ns falling edge needs at
    // least 7 samples for trapz() to get the switching loss right; 10 ns
    // only gives ~1.5 and underestimates Q1 loss badly.
    Simulate(
        SchFile,
        TxtFile,
        TimeStep  = 2n,
        TotalTime = 50m,
        PrintTime = 49.96m,
        PrintStep = 1,
        Rg        = Rg
    );

    // Coarse timestep for the waveform viewer. Nothing is integrated on
    // this file, so 10 ns is fine.
    Simulate(
        SchFile,
        SmvFile,
        TimeStep  = 10n,
        TotalTime = 50m,
        PrintTime = 49.96m,
        PrintStep = 1,
        Rg        = Rg
    );

    Row = CaseNo + "," + String(Vin_case) + "," + String(Io_target) + "," + String(Rload_case) + "," + String(Duty_case) + "," + String(Rg) + ",SEPIC_case_" + CaseNo + "_steady_state.txt";
    FileAppend(IndexFile, Row + _CRLF);

    i = i + 1;
}

Print("--- All SEPIC simulations done ---");
Print("9 .txt files and 9 .smv files should be in 'SEPIC NEW'.");
